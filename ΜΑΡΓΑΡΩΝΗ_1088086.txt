show_cell(N):- cell(N, Z), (Z==black, write('| b ');
                           Z==white, write('| w ');
                           Z==empty, write('| X ')).

show_puzzle:- nl, show_cell(1), show_cell(2), show_cell(3), show_cell(4), show_cell(5), show_cell(6), show_cell(7), nl.

move_left_black:- cell(X, empty), X>=2, X=<7, X1 is X-1, cell(X1, Z), Z==black,
retract(cell(X, empty)), assert(cell(X, black)), retract(cell(X1, black)), assert(cell(X1, empty)).

move_left_white:- cell(X, empty), X>=2, X=<7, X1 is X-1, cell(X1, Z), Z==white, retract(cell(X, empty)), assert(cell(X, white)), retract(cell(X1, white)), assert(cell(X1, empty)).
